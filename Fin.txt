function tarjetaLocalHtml(l, isPremium, esAdmin) {

  let abiertoBadge = '';
  let horarioMostrar = '';
  if(esAdmin) {
    let abierto = l.abierto !== undefined ? l.abierto : true;
    let horario = l.horario || '';
    let hoy = new Date().toISOString().split('T')[0];
    if(l.excepcionFecha === hoy) { if(l.excepcionCerrado) { abierto = false; horario = 'Hoy cerrado'; } else if(l.excepcionHorario) { horario = l.excepcionHorario; abierto = true; } }

  let bloqueoOverlay = bloqueado ? '<div class="local-bloqueado-overlay"><i class="fa fa-lock"></i> BLOQUEADO</div>' : '';
  return `<div class="${cardClass} relative" onclick="${bloqueado ? '' : `seleccionarLocal('${l.id}')`}">${bloqueoOverlay}${botonesEdicion}<div class="placeholder-img min-h-[100px]">${l.foto ? `<img src="${l.foto}" class="w-full h-full object-cover rounded">` : `<i class="fa fa-store text-4xl"></i>`}${btnAlquiler ? `<div class="absolute inset-0 flex items-center justify-center bg-black/60 rounded-lg"><span class="bg-green-600 text-white font-bold px-4 py-2 rounded-full">🟩 DISPONIBLE</span></div>` : ''}</div><h4 class="font-bold mt-2 text-center">${l.nombre}</h4><p class="text-gray-400 text-center text-sm">${l.rubro}</p><div class="flex justify-center gap-2 mt-1 flex-wrap">${abiertoBadge} ${estadoBadge}</div>${horarioMostrar}${l.fechaVencimiento && esAdmin ? `<p class="text-xs text-center text-gray-500 mt-1">📅 Vence: ${l.fechaVencimiento}</p>` : ''}</div>`;
}

function guardarLocal(id) {
  let nombre = document.getElementById('local-nombre').value.trim();
  let rubro = document.getElementById('local-rubro').value.trim();
  let categoria = document.getElementById('local-categoria').value;
  let fotoUrl = document.getElementById('local-foto-url').value.trim();
  let file = document.getElementById('local-foto-file').files[0];
  let horario = document.getElementById('local-horario').value.trim();
  let venc = document.getElementById('local-vencimiento').value;
  let centro = obtenerCentroActual();
  if(!nombre) { toast('❌ Nombre obligatorio'); return; }
  if(!id) { // Nuevo local: generar llave con documento y dueño
    let documento = document.getElementById('nuevo-documento')?.value.trim();
    let nuevoDueño = document.getElementById('nuevo-dueno')?.value.trim();
    if(!documento || !nuevoDueño) { toast('❌ Documento y nombre del dueño son obligatorios'); return; }
    let nuevaLlave = generarLlaveUnica(centro.nombre, documento);
    let nuevoLocal = { id: 'local_'+Date.now(), nombre, rubro, categoria, estado: 'AL_DIA', llave: nuevaLlave, dueño: nuevoDueño, foto: '', horario, abierto: true, fechaVencimiento: venc, productos: [] };
    if(file) { convertirImagenABase64(file, (base64) => { nuevoLocal.foto = base64; centro.locales.push(nuevoLocal); guardarDatos(); agregarHistorial(`🏪 Local agregado: ${nuevoLocal.nombre} con llave ${nuevaLlave}`); cerrarModal(); renderizarLocales(); toast(`✅ Local creado. Llave: ${nuevaLlave}`); }); }
    else { centro.locales.push(nuevoLocal); guardarDatos(); agregarHistorial(`🏪 Local agregado: ${nuevoLocal.nombre} con llave ${nuevaLlave}`); cerrarModal(); renderizarLocales(); toast(`✅ Local creado. Llave: ${nuevaLlave}`); }
  } else {
    // Edición: no regenerar llave
    let l = centro.locales.find(l => l.id === id);
    if(l) {
      l.nombre = nombre; l.rubro = rubro; l.categoria = categoria; l.horario = horario; l.fechaVencimiento = venc;
      if(file) convertirImagenABase64(file, (base64) => { l.foto = base64; guardarDatos(); agregarHistorial(`🏪 Local editado: ${nombre}`); cerrarModal(); renderizarLocales(); toast('✅ Local actualizado'); });
      else { guardarDatos(); agregarHistorial(`🏪 Local editado: ${nombre}`); cerrarModal(); renderizarLocales(); toast('✅ Local actualizado'); }
    }
  }
}

function generarInformeMASA() { let totalCentros=0, totalLocales=0, alDia=0, deuda=0, premium=0, estandar=0, ingresosAlquiler=0, ingresosPublicidad=0, espaciosOcupados=0, espaciosLibres=0; let localesList = []; for(let p of datos.paises) for(let c of p.ciudades) for(let ce of c.centros) { totalCentros++; if(ce.publicidad) for(let pub of ce.publicidad) { if(pub.activo) { espaciosOcupados++; ingresosPublicidad+=50000; } else espaciosLibres++; } for(let l of ce.locales) { totalLocales++; if(l.estado === 'AL_DIA') { alDia++; ingresosAlquiler += l.categoria==='PREMIUM'?50000:25000; } if(l.estado === 'DEUDA') deuda++; if(l.categoria === 'PREMIUM') premium++; else estandar++; let horarioMostrar = l.horario || ''; let abiertoHoy = l.abierto; let hoy = new Date().toISOString().split('T')[0]; if(l.excepcionFecha === hoy) { if(l.excepcionCerrado) { abiertoHoy=false; horarioMostrar='Hoy cerrado'; } else if(l.excepcionHorario) { horarioMostrar=l.excepcionHorario; abiertoHoy=true; } } localesList.push({ id: l.id, nombre: l.nombre, rubro: l.rubro, categoria: l.categoria, estado: l.estado, llave: l.llave||'', dueño: l.dueño||'', vencimiento: l.fechaVencimiento||'', foto: l.foto||'', horario: horarioMostrar, abierto: abiertoHoy, centro: ce.nombre, ciudad: c.nombre, pais: p.nombre }); } } let informe = { fecha: new Date().toISOString(), resumen: { totalCentros, totalLocales, alDia, deuda, premium, estandar, ingresosAlquiler, ingresosPublicidad, espaciosOcupados, espaciosLibres }, locales: localesList }; document.getElementById('informe-masa-texto').innerText = JSON.stringify(informe, null, 2); window.ultimoInformeMASA = informe; }
function copiarInformesMASA() { if(window.ultimoInformeMASA) { navigator.clipboard.writeText(JSON.stringify(window.ultimoInformeMASA,null,2)); toast('✅ Copiado'); } else toast('Generar informe primero'); }
function generarInforme() { let totalPaises = datos.paises.length, totalCiudades = 0, totalCentros = 0, totalLocales = 0, alDia = 0, deuda = 0, bloqueados = 0, vacantes = 0, premium = 0, estandar = 0; let vencimientosProximos = []; let hoy = new Date(); hoy.setHours(0,0,0,0); let en30Dias = new Date(hoy); en30Dias.setDate(en30Dias.getDate()+30); for(let p of datos.paises) for(let c of p.ciudades) { totalCiudades++; for(let ce of c.centros) { totalCentros++; for(let l of ce.locales) { totalLocales++; if(l.estado === 'AL_DIA') alDia++; else if(l.estado === 'DEUDA') deuda++; else if(l.estado === 'BLOQUEADO') bloqueados++; function buscarGlobal() {
  let term = document.getElementById('input-buscador').value.trim().toLowerCase();
  let resDiv = document.getElementById('resultados-busqueda');
  if(term.length < 2) { resDiv.innerHTML = ''; return; }
  let results = [];
  for(let p of datos.paises) for(let c of p.ciudades) for(let ce of c.centros) for(let l of ce.locales) {
    let puedeVer = false;
    if(usuarioActual?.tipo === 'admin') puedeVer = true;
    else if(usuarioActual?.tipo === 'local') puedeVer = (l.id === usuarioActual.localId);
    else puedeVer = (l.estado === 'AL_DIA' || l.estado === 'VACANTE');
    if(!puedeVer) continue;
    if(l.nombre.toLowerCase().includes(term) || (l.dueño||'').toLowerCase().includes(term) || (l.llave||'').toLowerCase().includes(term) || (l.rubro||'').toLowerCase().includes(term) || ce.nombre.toLowerCase().includes(term) || c.nombre.toLowerCase().includes(term) || p.nombre.toLowerCase().includes(term)) {
      results.push({ ...l, nombrePais: p.nombre, nombreCiudad: c.nombre, nombreCentro: ce.nombre, paisId: p.id, ciudadId: c.id, centroId: ce.id });
    }
  }
  resDiv.innerHTML = results.slice(0, 15).map(r => `<div class="bg-gray-800 p-2 rounded-lg cursor-pointer hover:bg-gray-700" onclick="irAProductoDesdeBusqueda('${r.paisId}','${r.ciudadId}','${r.centroId}','${r.id}')"><p class="font-semibold">${r.nombre}</p><p class="text-xs text-gray-400">${r.nombreCentro} · ${r.nombreCiudad}</p><span class="text-oro text-xs">${r.rubro}</span></div>`).join('');
}

}
